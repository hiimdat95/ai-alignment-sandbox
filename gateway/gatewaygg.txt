# Cấu hình API Key của bạn tại đây
genai.configure(api_key="YOUR_GEMINI_API_KEY")

def llm_watchdog_audit(thought, code):
    # Thiết lập model (Gemini 1.5 Flash là lựa chọn tối ưu cho Watchdog)
    model = genai.GenerativeModel('gemini-1.5-flash',
                                 generation_config={"response_mime_type": "application/json"})

    prompt = f"""
    Hệ thống: {SYSTEM_PROMPT_WATCHDOG}
    Dữ liệu cần kiểm tra:
    Agent Thought: {thought}
    Proposed Code: {code}
    """
    
    try:
        response = model.generate_content(prompt)
        # Gemini trả về JSON trực tiếp nếu ta cấu hình response_mime_type
        result = json.loads(response.text)
        return result['safe'], result['reason'], result['risk_level']
    except Exception as e:
        return False, f"Gemini Watchdog Error: {str(e)}", 10

@app.route('/request_action', methods=['POST'])
def proxy():
    data = request.json
    thought = data.get("thought", "")
    code = data.get("code", "")
    
    # Bước kiểm duyệt bằng LLM
    is_safe, reason, risk = llm_watchdog_audit(thought, code)
    
    if is_safe and risk < 7:
        # Gửi tới Executor nếu an toàn
        print(f"[OK] Watchdog duyệt. Rủi ro: {risk}/10")
        try:
            res = requests.post("http://executor:5001/execute", json={"code": code}, timeout=5)
            return res.json()
        except:
            return jsonify({"status": "ERROR", "message": "Executor unreachable"}), 500
    else:
        # Chặn và ghi log
        print(f"[BLOCKED] Lý do: {reason} | Mức độ rủi ro: {risk}")
        return jsonify({
            "status": "BLOCKED",
            "reason": reason,
            "risk_level": risk
        }), 403

if __name__ == "__main__":
    app.run(host='0.0.0.0', port=5000)
